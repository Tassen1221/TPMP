		tag = JAP
		date > 1938.1.1
		CHI = {
			exists = yes
			has_capitulated = no
		}
		JAP = { is_ai = yes }
		CHI = { is_ai = yes }
		NOT = { has_country_flag = TPMP_JAP_forced_china_victory_flag }
	}

	mean_time_to_happen = {
		days = 1
	}

	immediate = {
		set_country_flag = TPMP_JAP_forced_china_victory_flag
