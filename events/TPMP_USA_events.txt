country_event = {
    id = TPMP_USA_military.1

    hidden = yes
    
    immediate = {
        IF = {
			limit = {
				NOT = {
					has_idea = TPMP_USA_iterative_tank_design_idea_3
				}
			}
			country_event = {
				id = TPMP_USA_military.1
				days = 180
			}
		}
		IF = {
			limit = {
				has_idea = TPMP_USA_iterative_tank_design_idea_3
			}
			swap_ideas = {
				remove_idea = TPMP_USA_iterative_tank_design_idea_3
				add_idea = TPMP_USA_iterative_tank_design_idea_4
			}
		}
		ELSE_IF = {
			limit = {
				has_idea = TPMP_USA_iterative_tank_design_idea_2
			}
			swap_ideas = {
				remove_idea = TPMP_USA_iterative_tank_design_idea_2
				add_idea = TPMP_USA_iterative_tank_design_idea_3
			}
		}
		ELSE_IF = {
			limit = {
				has_idea = TPMP_USA_iterative_tank_design_idea_1
			}
			swap_ideas = {
				remove_idea = TPMP_USA_iterative_tank_design_idea_1
				add_idea = TPMP_USA_iterative_tank_design_idea_2
			}
		}
		ELSE_IF = {
			limit = {
				has_idea = TPMP_USA_iterative_tank_design_idea_0
			}
			swap_ideas = {
				remove_idea = TPMP_USA_iterative_tank_design_idea_0
				add_idea = TPMP_USA_iterative_tank_design_idea_1
			}
		}
    }
}
