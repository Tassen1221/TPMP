		IF = {
			limit = {
				NOT = {
					has_dynamic_modifier = {
						modifier = TPMP_USA_great_depression_modifier
					}
				}
			}
			add_dynamic_modifier = {
				modifier = TPMP_USA_great_depression_modifier
			}
		}
		custom_effect_tooltip = USA_decrease_great_depression_tt

		}
		add_to_variable = {
			TPMP_USA_great_depression_stab_var = -0.075
			tooltip = stability_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_cons_var = -0.03
			tooltip = production_speed_buildings_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_eff_cap_var = -0.03
			tooltip = production_factory_max_efficiency_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_eff_gain_var = -0.03
			tooltip = production_factory_efficiency_gain_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_factory_var = -0.03
			tooltip = industrial_capacity_factory_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_dock_var = -0.06
			tooltip = industrial_capacity_dockyard_tt
		}
		hidden_effect = {
			set_variable = {
				TPMP_USA_roosevelt_recession_count_var = 3
