# TPMP MIOs of SOV. Each is a country copy of a generic archetype (include) with its own name and icon.
# The trait trees live in the TPMP_generic_*_organizations.txt files.

### Tanks ###

SOV_morozov_design_bureau_organization = {
	include = TPMP_generic_standard_tank_organization
	icon = GFX_idea_morozov_design_bureau

	allowed = {
		original_tag = SOV
	}

	# Country-unique traits (gold outline)
	add_trait = {
		token = SOV_mio_trait_merged_tank_and_materiel_plants
		name = SOV_mio_trait_merged_tank_and_materiel_plants
		icon = GFX_generic_mio_trait_icon_build_cost_ic
		special_trait_background = yes

		position = { x=1 y=0 }

		available = {
			hidden_trigger = { NOT = { has_mio_flag = TPMP_MIO_selected_flag } }
			FROM = { has_completed_focus = SOV_merge_tank_and_materiel_plants }
			hidden_trigger = { FROM = { NOT = { has_country_flag = SOV_merged_tank_and_materiel_plants_flag } } }
		}

		on_complete = {
			custom_effect_tooltip = TPMP_MIO_selected_tt
			hidden_effect = {
				FROM = {
					every_military_industrial_organization = {
						limit = { TPMP_MIO_is_tank_category = yes }
						set_mio_flag = TPMP_MIO_selected_flag
					}
				}
				clr_mio_flag = TPMP_MIO_selected_flag
			}
			custom_effect_tooltip = generic_skip_one_line_tt
			custom_effect_tooltip = generic_trait_will_not_be_available_in_other_organizations
			FROM = { set_country_flag = SOV_merged_tank_and_materiel_plants_flag }
		}
		equipment_bonus = {
			build_cost_ic = -0.05
		}
	}
}
