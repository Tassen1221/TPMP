ENG_supermarine_organization = {
	include = TPMP_generic_agility_aircraft_organization
	icon = GFX_idea_supermarine

	allowed = {
		original_tag = ENG
	}

	# Country-unique traits (gold outline)
	add_trait = {
		token = ENG_mio_trait_stressed_skin_construction
		name = ENG_mio_trait_stressed_skin_construction
		icon = GFX_generic_mio_department_icon_plane_fighter_materials
		special_trait_background = yes

		position = { x=1 y=1 }
		relative_position_id = generic_mio_trait_agility_aircraft_2c
		any_parent = { generic_mio_trait_agility_aircraft_2c }

		available = {
			hidden_trigger = { NOT = { has_mio_flag = TPMP_MIO_selected_flag } }
		}

		equipment_bonus = {
			air_agility = 0.1
		}
	}
	add_trait = {
		token = ENG_mio_trait_elliptical_wing
		name = ENG_mio_trait_elliptical_wing
		icon = GFX_generic_mio_trait_icon_air_defence
		special_trait_background = yes

		position = { x=0 y=1 }
		relative_position_id = ENG_mio_trait_stressed_skin_construction
		any_parent = { ENG_mio_trait_stressed_skin_construction }

		available = {
			hidden_trigger = { NOT = { has_mio_flag = TPMP_MIO_selected_flag } }
		}

		equipment_bonus = {
			air_defence = 0.1
		}
	}
}
