		available = {
			hidden_trigger = { NOT = { has_mio_flag = TPMP_MIO_selected_flag } }

### Aircraft ###

GER_messerschmitt_organization = {
	include = TPMP_generic_agility_aircraft_organization
	icon = GFX_idea_messerschmitt

	allowed = {
		original_tag = GER
	}

	# Country-unique traits (gold outline)
	add_trait = {
		token = GER_mio_trait_extended_service_life
		name = GER_mio_trait_extended_service_life
		icon = GFX_generic_mio_department_icon_plane_fighter_production
		special_trait_background = yes

		position = { x=-3 y=0 }
		relative_position_id = generic_mio_trait_agility_aircraft_1a

		available = {
			hidden_trigger = { NOT = { has_mio_flag = TPMP_MIO_selected_flag } }
		}

		on_complete = {
			custom_effect_tooltip = TPMP_MIO_selected_tt
			hidden_effect = {
				FROM = {
					every_military_industrial_organization = {
						limit = { TPMP_MIO_is_aircraft_category = yes }
						set_mio_flag = TPMP_MIO_selected_flag
					}
				}
				clr_mio_flag = TPMP_MIO_selected_flag
			}
		}
		equipment_bonus = {
			build_cost_ic = -0.05
		}
	}
}
