		}

			has_political_power > 49

		}
		custom_cost_text = decision_cost_CP_50_pp_50
        icon = GFX_decision_generic_civil_support
		fire_only_once = yes
        complete_effect = {
			every_state = {
				limit = {
					controller = {
						OR = {
							tag = GER
							is_ally_with = GER
						}
					}
					any_neighbor_state = {
						controller = {
							OR = {
								tag = SOV
								is_ally_with = SOV
							}
						}
					}
				}
				add_dynamic_modifier = {
					modifier = TPMP_hold_our_ground_modifier
					scope = GER
					days = 450
				}
				hidden_effect = {
					add_dynamic_modifier = {
						modifier = TPMP_hold_our_ground_modifier_hidden
						scope = ITA
						days = 450
					}
