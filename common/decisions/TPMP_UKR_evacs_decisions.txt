	TPMP_EvacState_00_mission = {

		activation = {
			always = no
		}

		allowed = {
   			original_tag = UKR
		}

		available = {
			always = no
		}

		visible = {
    		always = yes
		}

		icon = GFX_decision_generic_factory
		days_mission_timeout = global.var_tpmp_EvacStateDuration
   		selectable_mission = no
		is_good = no

		timeout_effect = {
			hidden_effect = {
				UKR = {
					
					every_owned_state = {
						limit = {
							is_in_array = { global.array_tpmp_EvacState_00 = THIS }
						}
						if = {
							limit = {
								THIS = {
									OR = {
										industrial_complex > 0
										arms_factory > 0
									}
								}
							}

							set_variable = { THIS:var_tpmp_EvacStateAmountScope = global.var_tpmp_EvacStateAmount }
							

							
							tpmp_evac_calculate_evacuate_factories = yes


							#log = "[THIS.GetID] is NOT empty"
								
							}
						else = {
							#log = "[THIS.GetID] is empty"
							set_state_flag = flag_tpmp_state_is_evacuated
						}
					}

					if = {
						limit = {
							all_of_scopes = {
								array =  global.array_tpmp_EvacState_00
								OR = {
									has_state_flag = flag_tpmp_state_is_evacuated
									NOT = {
										is_controlled_by = UKR
									}
								}

							}
						}
						#log = "All States are evaced"
						tpmp_evac_calculate_evac_days = yes
						activate_mission = TPMP_EvacState_01_mission
					}
					else = {
						#log = "All state ARE NOT evaced"
						activate_mission = TPMP_EvacState_00_mission

						tpmp_evac_calculate_factory_return_civ = yes
						tpmp_evac_calculate_factory_return_mil = yes
					}

					every_owned_state = { #Todo Question of optimization. If it is evaced, is it not better to do limit = in every owned state? Possible problem - building a mil after evaced doesn't let it evac
						clr_state_flag = flag_tpmp_state_is_evacuated
					}
				}
			}			
		}
	}
