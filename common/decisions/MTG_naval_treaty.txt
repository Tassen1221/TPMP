		icon = generic
			
	
		available = {
			
		}
	
		cost = 50
		fire_only_once = yes
		ai_will_do = {
			factor = 3
			modifier = {
				tag = JAP
				factor = 5 #The Yamato is a 35,000 ton ship what are you talking about
			}
		}
	
		visible = {
			OR = {
				has_idea = MTG_naval_treaty_adherent
				has_idea = MTG_naval_treaty_adherent_reduced
			}
			OR = {
				has_government = fascism
				has_government = communism
				# Neutral Japan walks out cheaply too, so it deceives at the same price rather than the democratic one
				AND = {
					original_tag = JAP
					has_government = neutrality
				}
			}
		}
	
		complete_effect = {
			if = {
				limit = {
					has_idea = MTG_naval_treaty_adherent
				}
				swap_ideas = {
					remove_idea = MTG_naval_treaty_adherent
					add_idea = MTG_naval_treaty_cheating
				}
			}
			if = {
				limit = {
					has_idea = MTG_naval_treaty_adherent_reduced
				}
				swap_ideas = {
					remove_idea = MTG_naval_treaty_adherent_reduced
					add_idea = MTG_naval_treaty_cheating_reduced
				}
			}
			# Dates the deception, which is what lets it surface on its own where there is no intelligence work to catch it
			set_country_flag = MTG_naval_treaty_cheating_since
			custom_effect_tooltip = MTG_cheat_on_treaty_risk_tt
		}
	}

			OR = {
				AND = {
					has_dlc = "La Resistance"
					intel_level_over = {
						target = FROM
						navy_intel > 0.65
					}
				}
				# Without the intelligence systems nothing can build a case early, so the deception only surfaces with time
				AND = {
					NOT = { has_dlc = "La Resistance" }
					FROM = {
						has_country_flag = {
							flag = MTG_naval_treaty_cheating_since
							days > 420
						}
					}
				}
			}
		}

		cost = 75
		ai_will_do = {
			factor = 10
			modifier = {
				NOT = {

					}

				factor = 0
			}
		}

		visible = {
			has_naval_treaty_trigger = yes
		}

		complete_effect = {
			custom_effect_tooltip = MTG_expose_FROM_treaty_violation_tt
			add_named_threat = { threat = 1 name = MTG_expose_FROM_treaty_violation_threat }
			FROM = {
				set_country_flag = MTG_treaty_violation_exposed
				add_political_power = -50
				add_stability = -0.05
				if = {
					limit = {
						OR = {
							has_government = democratic
							has_government = neutrality
						}
					}
					add_stability = -0.05
					add_war_support = -0.05
				}
				# PREV is the violator in here, so every remaining member turns against them

					limit = {
						has_naval_treaty_trigger = yes
					}
					add_opinion_modifier = {
						target = PREV
						modifier = broken_naval_treaty_opinion_mod
					}
				}
				if = {
					limit = {
						NOT = {
							any_other_country = {

									has_idea = MTG_naval_treaty_cheating_reduced
