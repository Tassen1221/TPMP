        }

		days_remove = 180

		cost = GER_war_bonds_var?50

		modifier = {
			consumer_goods_factor = -0.15
		}
		complete_effect = {
            IF = {
                limit = {

                        TPMP_USA_deficit_cg_var > 0
                    }

			IF = {
				limit = {
					NOT = {

					var = GER_war_bonds_var
					value = 50
				}
			}

				limit = {
					has_variable = GER_war_bonds_war_support_var
				}
				add_war_support = GER_war_bonds_war_support_var
			}
			add_to_variable = {
				var = GER_war_bonds_war_support_var
				value = -0.025
			}

				var = GER_war_bonds_var

			}
		}
