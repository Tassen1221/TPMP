### Mobilization Law ###

TPMP_ROM_ai_save_for_partial = {
	allowed = {
		original_tag = ROM
	}
	enable = {
		has_war_support > 0.25
		OR = {
			NOT = { has_idea = partial_economic_mobilisation }
			NOT = { has_idea = war_economy }
			NOT = { has_idea = tot_economic_mobilisation }
		}
	}
	abort_when_not_enabled = yes

	ai_strategy = {
		type = pp_spend_amount
		id = idea
		value = 500
	}
}
