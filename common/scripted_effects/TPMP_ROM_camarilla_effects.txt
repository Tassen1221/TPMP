TPMP_ROM_camarilla_start_effect = {
	set_variable = { TPMP_ROM_camarilla_index = 0 }
	set_country_flag = TPMP_ROM_camarilla_active
	add_ideas = TPMP_ROM_camarilla_titulescu
	activate_mission = TPMP_ROM_camarilla_mission_0
}

TPMP_ROM_camarilla_advance_0_to_1 = {
	remove_ideas = TPMP_ROM_camarilla_titulescu
	if = {
		limit = { has_completed_focus = TPMP_ROM_camarilla_renewed }
		add_ideas = TPMP_ROM_camarilla_lupescu_renewed
	}
	else = {
		add_ideas = TPMP_ROM_camarilla_lupescu
	}
	set_variable = { TPMP_ROM_camarilla_index = 1 }
	activate_mission = TPMP_ROM_camarilla_mission_1
}

TPMP_ROM_camarilla_prolong_effect = {
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 0 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_0 days = 28 }
	}
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 1 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_1 days = 28 }
	}
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 2 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_2 days = 28 }
	}
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 3 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_3 days = 28 }
	}
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 4 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_4 days = 28 }
	}
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 5 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_5 days = 28 }
	}
	if = {
		limit = { check_variable = { TPMP_ROM_camarilla_index = 6 } }
		add_days_mission_timeout = { mission = TPMP_ROM_camarilla_mission_6 days = 28 }
	}
}
