recalculate_congressional_ui = {
	for_loop_effect = {
		start = 0
		end = USA.left_wing_democrat_support_var
		set_variable = {
			USA.congress_frame_array^v = 1
		}
	}
	set_temp_variable = { next_end = USA.left_wing_democrat_support_var }
	add_to_temp_variable = { next_end = USA.new_deal_democrat_support_var }
	for_loop_effect = {
		start = USA.left_wing_democrat_support_var
		end = next_end
		set_variable = {
			USA.congress_frame_array^v = 2
		}
	}
	set_temp_variable = { next_start = next_end }
	add_to_temp_variable = { next_end = USA.conservative_democrat_support_var }
	for_loop_effect = {
		start = next_start
		end = next_end
		set_variable = {
			USA.congress_frame_array^v = 3
		}
	}
	set_temp_variable = { next_start = next_end }
	add_to_temp_variable = { next_end = USA.progressive_republican_support_var }
	for_loop_effect = {
		start = next_start
		end = next_end
		set_variable = {
			USA.congress_frame_array^v = 4
		}
	}
	set_temp_variable = { next_start = next_end }
	add_to_temp_variable = { next_end = USA.business_republican_support_var }
	for_loop_effect = {
		start = next_start
		end = next_end

			USA.congress_frame_array^v = 5

			USA.congress_frame_array^v = 6
		}
	}
}

USA_decrease_great_depression = {
	IF = {
		limit = {
			has_dynamic_modifier = {
				modifier = TPMP_USA_great_depression_modifier
			}
		}
		custom_effect_tooltip = USA_decrease_great_depression_tt
		add_to_variable = {
			TPMP_USA_great_depression_cg_var = -0.05
			tooltip = consumer_goods_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_stab_var = 0.025
			tooltip = stability_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_cons_var = 0.01
			tooltip = production_speed_buildings_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_eff_cap_var = 0.01
			tooltip = production_factory_max_efficiency_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_eff_gain_var = 0.01
			tooltip = production_factory_efficiency_gain_factor_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_factory_var = 0.01
			tooltip = industrial_capacity_factory_tt
		}
		add_to_variable = {
			TPMP_USA_great_depression_dock_var = 0.02
			tooltip = industrial_capacity_dockyard_tt
		}
	}
	ELSE = {
		custom_effect_tooltip = USA_decrease_great_depression_fail_tt
	}
	hidden_effect = {
		IF = {
			limit = {
				check_variable = {
					var = TPMP_USA_great_depression_cg_var
					value = 0
					compare = less_than_or_equals
				}
			}
			remove_dynamic_modifier = {
				modifier = TPMP_USA_great_depression_modifier
			}
		}
	}
}
