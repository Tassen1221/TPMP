recalculate_congressional_ui = {
	for_loop_effect = {
		start = 0
		end = USA.left_wing_democrat_support_var
		set_variable = {
			USA.congress_frame_array^v = 1
		}
	}
	set_temp_variable = { next_end = USA.left_wing_democrat_support_var }
	add_to_temp_variable = { next_end = USA.new_deal_democrat_support_var }
	for_loop_effect = {
		start = USA.left_wing_democrat_support_var
		end = next_end
		set_variable = {
			USA.congress_frame_array^v = 2
		}
	}
	set_temp_variable = { next_start = next_end }
	add_to_temp_variable = { next_end = USA.conservative_democrat_support_var }
	for_loop_effect = {
		start = next_start
		end = next_end

			USA.congress_frame_array^v = 3

	}

		start = next_start
