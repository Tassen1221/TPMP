TPMP_FRA_end_metropolitan_campaign_effect = {
        if = { limit = { has_war_with = GER NOT = { has_country_flag = TPMP_FRA_capitulated } }
            if = { limit = { NOT = { has_country_flag = TPMP_FRA_deadline_announced } }
                set_country_flag = TPMP_FRA_deadline_announced
                country_event = { id = tpmp_france.5 hours = 1 }
            }
            remove_ideas = TPMP_FRA_collapse_june
            remove_ideas = TPMP_FRA_collapse_august
            add_ideas = TPMP_FRA_collapse_terminal
            every_owned_state = {
                limit = { is_core_of = ROOT is_on_continent = europe }
                set_state_controller_to = GER
            }
        }
    }

TPMP_FRA_campaign_tick_effect = {
        TPMP_FRA_initialize_effect = yes
        if = { limit = { has_capitulated = yes NOT = { has_country_flag = TPMP_FRA_capitulated } } TPMP_FRA_enter_exile_effect = yes }
        if = { limit = { has_country_flag = TPMP_FRA_capitulated }
            # Late vanilla armistice events can add defeatism after the immediate hook.
            remove_ideas = FRA_defeatism
            remove_ideas = TPMP_FRA_collapse_terminal
        }
        else_if = { limit = { has_war_with = GER }
            if = { limit = { date > 1940.9.30 } TPMP_FRA_end_metropolitan_campaign_effect = yes }
            else_if = { limit = { date > 1940.7.31 }
                remove_ideas = TPMP_FRA_collapse_june
                if = { limit = { NOT = { has_idea = TPMP_FRA_collapse_august } } add_ideas = TPMP_FRA_collapse_august country_event = { id = tpmp_france.4 hours = 1 } }
            }
            else_if = { limit = { date > 1940.5.31 }
                if = { limit = { NOT = { has_idea = TPMP_FRA_collapse_june } } add_ideas = TPMP_FRA_collapse_june country_event = { id = tpmp_france.3 hours = 1 } }
            }
        }
        else = {
            remove_ideas = TPMP_FRA_collapse_june remove_ideas = TPMP_FRA_collapse_august remove_ideas = TPMP_FRA_collapse_terminal
        }
    }
