# Strategic Situations panel - scope: the player's country

TPMP_strat_open_window = {
	set_country_flag = TPMP_strat_window_open
	# First time the panel is opened: start on the Eastern Front
	if = {
		limit = {
			NOT = { check_variable = { TPMP_strat_tab > 0 } }
		}
		set_variable = { TPMP_strat_tab = 1 }
	}
	TPMP_strat_refresh_tabs = yes
}

# Tab sprite frames: 1 = selected, 2 = idle
TPMP_strat_refresh_tabs = {
	set_variable = { TPMP_strat_tab_1_frame = 2 }
	set_variable = { TPMP_strat_tab_2_frame = 2 }
	set_variable = { TPMP_strat_tab_3_frame = 2 }
	set_variable = { TPMP_strat_tab_4_frame = 2 }
	if = {
		limit = { check_variable = { TPMP_strat_tab = 2 } }
		set_variable = { TPMP_strat_tab_2_frame = 1 }
	}
	else_if = {
		limit = { check_variable = { TPMP_strat_tab = 3 } }
		set_variable = { TPMP_strat_tab_3_frame = 1 }
	}
	else_if = {
		limit = { check_variable = { TPMP_strat_tab = 4 } }
		set_variable = { TPMP_strat_tab_4_frame = 1 }
	}
	else = {
		set_variable = { TPMP_strat_tab_1_frame = 1 }
	}
}
