#UKR factory evac scripted effects

tpmp_evac_calculate_evacuate_factories = {
	THIS = {
		set_variable = { global.var_tpmp_EvacState_mil_count = arms_factory_level }
		set_variable = { global.var_tpmp_EvacState_civ_count = industrial_complex_level }	
	}
	
	if = {
		limit = {
			THIS = {
				industrial_complex > 0
			}
		}

		#Calculating Civs

		set_temp_variable = { var_temp_tpmp_Evac_civ_count = var_tpmp_EvacStateAmountScope }
		if = {
			limit = {
				check_variable = {
					var = global.var_tpmp_EvacState_civ_count
					value = var_tpmp_EvacStateAmountScope
					compare = less_than_or_equals
				}
			}
			set_temp_variable = { var_temp_tpmp_Evac_civ_count = global.var_tpmp_EvacState_civ_count }
		}

		#Removing Civs

		remove_building = {
			type = industrial_complex
			level = THIS:var_temp_tpmp_Evac_civ_count
		}
		#log = "Removed [?var_temp_tpmp_Evac_civ_count] civs from [THIS.GetID]"


		UKR = {
			add_offsite_building = { type = industrial_complex level = THIS:var_temp_tpmp_Evac_civ_count }
		}

		#Civs Counter
		add_to_variable = { global.var_tpmp_evaced_civ_factories = var_temp_tpmp_Evac_civ_count }

		#log = "      Total UKR evaced civs: [?global.var_tpmp_evaced_civ_factories]"		
	}

	if = {
		limit = {
			THIS = {
				arms_factory > 0
			}
		}

		#Calculating Mils

		set_temp_variable = { var_temp_tpmp_Evac_mil_count = var_tpmp_EvacStateAmountScope }
		if = {
			limit = {
				check_variable = {
					var = global.var_tpmp_EvacState_mil_count
					value = var_tpmp_EvacStateAmountScope
					compare = less_than_or_equals
				}
			}
			set_temp_variable = { var_temp_tpmp_Evac_mil_count = global.var_tpmp_EvacState_mil_count }
		}		

		#Removing mils

		remove_building = {
			type = arms_factory
			level = THIS:var_temp_tpmp_Evac_mil_count
		}
		#log = "Removed [?var_temp_tpmp_Evac_mil_count] mils from [THIS.GetID]"

		UKR = {
			add_offsite_building = { type = arms_factory level = THIS:var_temp_tpmp_Evac_civ_count }
		}

		#Mils Counter
		add_to_variable = { global.var_tpmp_evaced_mil_factories = var_temp_tpmp_Evac_mil_count }

		#log = "      Total UKR evaced mils: [?global.var_tpmp_evaced_mil_factories]"		
	}
}

# Return calculation

tpmp_evac_calculate_factory_return_mil = { #to avoid unnecessary checks and keep a way to have all the calculations in 1 place, we'll spit them for factories evaced

	#Get Sov mils

	SOV = {
		set_variable = { global.var_tpmp_SOV_mil_factories = num_of_military_factories }
	}

	#log = "    Total sov mils: [?global.var_tpmp_SOV_mil_factories]"
	
	#Calculator

	set_temp_variable = { var_temp_tpmp_Evac_calculator = global.var_tpmp_evaced_mil_factories }
	divide_temp_variable = { var_temp_tpmp_Evac_calculator = global.var_tpmp_SOV_mil_factories }
