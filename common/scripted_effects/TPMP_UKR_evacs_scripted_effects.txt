tpmp_evac_calculate_evacuate_factories = {
	THIS = {
		set_variable = { global.var_tpmp_EvacState_mil_count = arms_factory_level }
		set_variable = { global.var_tpmp_EvacState_civ_count = industrial_complex_level }	
	}

	if = {
		limit = {
			check_variable = { global.var_tpmp_EvacState_civ_count > var_tpmp_EvacStateAmountScope }
		}
		remove_building = {
			type = industrial_complex
    		level = THIS:var_tpmp_EvacStateAmountScope #Todo add here a variable
		}
		#log = "Removed [?var_tpmp_EvacStateAmountScope] civs from [THIS.GetID]"	
		SOV = {
			random_owned_controlled_state = {
				limit = {
					is_evac_SOV_state = yes

					free_building_slots = {
						building = industrial_complex
						size > 2
						include_locked = no
					}
				}
				add_building_construction = {
					type = industrial_complex
					level = THIS:var_tpmp_EvacStateAmountScope
					instant_build = yes
				}
				#log = "[THIS.GetName] Added [?var_tpmp_EvacStateAmountScope] civs to [THIS.GetID]"
				tpmp_evac_calculate_factory_return_civ_3 = yes
			}
		}
	}
	else = {
		if = {
			limit = {
				check_variable = { global.var_tpmp_EvacState_civ_count > 1 }
				check_variable = { global.var_tpmp_EvacStateAmount > 1 }
			}
			remove_building = {
				type = industrial_complex
				level = 2
			}
			#log = "Removed 2 civ from [THIS.GetID]"
			SOV = {
				random_owned_controlled_state = {
					limit = {
						is_evac_SOV_state = yes

						free_building_slots = {
							building = industrial_complex
							size > 1
							include_locked = no
						}
					}
					add_building_construction = {
						type = industrial_complex
						level = 2
						instant_build = yes
					}
					#log = "[THIS.GetName] Added 2 civs to [THIS.GetID]"
					tpmp_evac_calculate_factory_return_civ_2 = yes				
				}
			}			
		}
		else_if = {
			limit = {
				check_variable = { global.var_tpmp_EvacState_civ_count > 0 }				
			}
			remove_building = {
				type = industrial_complex
				level = 1
			}
			#log = "Removed 1 civ from [THIS.GetID]"
			SOV = {
				random_owned_controlled_state = {
					limit = {
						is_evac_SOV_state = yes

						free_building_slots = {
							building = industrial_complex
							size > 0
							include_locked = no
						}
					}
					add_building_construction = {
						type = industrial_complex
						level = 1
						instant_build = yes
					}
					#log = "[THIS.GetName] Added 1 civs to [THIS.GetID]"
					tpmp_evac_calculate_factory_return_civ_1 = yes				
				}
			}			
		}
	}

	if = {
		limit = {
			check_variable = { global.var_tpmp_EvacState_mil_count > var_tpmp_EvacStateAmountScope }
		}
		remove_building = {
			type = arms_factory
    		level = THIS:var_tpmp_EvacStateAmountScope
		}
		#log = "Removed [?var_tpmp_EvacStateAmountScope] mils from [THIS.GetID]"	
		SOV = {
			random_owned_controlled_state = {
				limit = {
					is_evac_SOV_state = yes

					free_building_slots = {
						building = arms_factory
						size > 2
						include_locked = no
					}
				}
				add_building_construction = {
					type = arms_factory
					level = THIS:var_tpmp_EvacStateAmountScope
					instant_build = yes
				}
				#log = "[THIS.GetName] Added [?var_tpmp_EvacStateAmountScope] mils to [THIS.GetID]"
				tpmp_evac_calculate_factory_return_mil_3 = yes
			}
		}
	}
	else = {
		if = {
			limit = {
				check_variable = { global.var_tpmp_EvacState_mil_count > 1 }
				check_variable = { global.var_tpmp_EvacStateAmount > 1 }
			}
			remove_building = {
				type = arms_factory
				level = 2
			}
			#log = "Removed 2 mil from [THIS.GetID]"
			SOV = {
				random_owned_controlled_state = {
					limit = {
						is_evac_SOV_state = yes

						free_building_slots = {
							building = arms_factory
							size > 1
							include_locked = no
						}
					}
					add_building_construction = {
						type = arms_factory
						level = 2
						instant_build = yes
					}
					#log = "[THIS.GetName] Added 2 mils to [THIS.GetID]"
					tpmp_evac_calculate_factory_return_mil_2 = yes				
				}
			}			
		}
		else_if = {
			limit = {
				check_variable = { global.var_tpmp_EvacState_mil_count > 0 }				
			}
			remove_building = {
				type = arms_factory
				level = 1
			}
			#log = "Removed 1 mil from [THIS.GetID]"
			SOV = {
				random_owned_controlled_state = {
					limit = {
						is_evac_SOV_state = yes

						free_building_slots = {
							building = arms_factory
							size > 0
							include_locked = no
						}
					}
					add_building_construction = {
						type = arms_factory
						level = 1
						instant_build = yes
					}
					#log = "[THIS.GetName] Added 1 mils to [THIS.GetID]"
					tpmp_evac_calculate_factory_return_mil_1 = yes				
				}
			}			
		}
	}

}

}

	set_variable = { global.var_tpmp_Evac_factory_return_mic = 0 } 
}
