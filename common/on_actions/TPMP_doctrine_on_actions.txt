						}
					} 
					NOT = { 
						has_state_flag = {
						    flag = para_sabotage_effect
						    value = 0
						}
					}
					CONTROLLER = { NOT = { tag = FROM } }
				}
				damage_building = {
					type = industrial_complex
					damage = 3
				}
				damage_building = {
					type = arms_factory
					damage = 3
				}
				set_state_flag = {
					flag = para_sabotage_effect
					days = 180
				}
			}
			if = {
				limit = { 
					FROM = {
						has_mastery_level = {
							amount = 5
							sub_doctrine = special_forces_doctrine_2
						}
					} 
					NOT = { 
						has_state_flag = {
						    flag = para_combat_effect
						    value = 0
						}
					}
				}
				every_state_division = {
					limit = { 
						OWNER = { has_war_with = FROM }
					}
					set_unit_organization = 0.75
				}
				set_state_flag = {
					flag = para_combat_effect
					days = 60
				}
			}
		}
	}
