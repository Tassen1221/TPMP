on_actions = {
	on_startup = {
		effect = {
			every_country = {
				set_variable = {
					fleet_limit_var = 100
				}
				add_dynamic_modifier = {
					modifier = TPMP_naval_penalty
				}
			}
		}
	}
	on_daily = {
		effect = {
			calculate_total_ship_num = yes
			every_navy_leader = {
				limit = {
					has_trait = TPMP_reserve_fleet_trait
				}
				calculate_reserve_ship_num = yes
			}
			#Calculating the active fleet size
			calculate_fleet_size = yes

			#Displaying active fleet size for topbar element
			set_variable = { active_fleet_var = total_ship_var }

			#Setting the constants for calculating the fleet limit penalty
			set_temp_variable = { fleet_penalty_factor_a_var = 0.25 }
			multiply_temp_variable = { fleet_penalty_factor_a_var = fleet_limit_var }
			set_temp_variable = { fleet_penalty_factor_b_var = 0.5 }
			multiply_temp_variable = { fleet_penalty_factor_b_var = fleet_limit_var }
			#Calculating the fleet limit penalty
			subtract_from_variable = { total_ship_var = fleet_penalty_factor_b_var }
			divide_temp_variable = { fleet_penalty_factor_a_var = total_ship_var }
			set_variable = { total_ship_var = fleet_penalty_factor_a_var }
			subtract_from_variable = { total_ship_var = 0.5 }

			#Setting the penalty for exceeding the fleet limit dynamic modifier
			IF = {
				check_variable = {
					active_fleet_var < fleet_limit_var
				}
				set_variable = { fleet_limit_penalty_var = 0 }
			}
			ELSE = {
				set_variable = { fleet_limit_penalty_var = total_ship_var }
			}
			clamp_variable = {
				var = fleet_limit_penalty_var
				min = -0.5
				max = 0
			}
		}
	}
}
