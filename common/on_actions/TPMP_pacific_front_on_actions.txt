	on_declare_war = {
		effect = {
			if = {
				limit = {
					tag = JAP

						OR = {
							tag = USA

					}
				}
				JAP = {
					add_ideas = TPMP_JAP_pacific_front_debuff_idea_10
					hidden_effect = {
						add_ideas = TPMP_JAP_formosa_idea
						add_ideas = TPMP_JAP_okinawa_idea
					}
				}
				USA = {
					add_ideas = TPMP_USA_pacific_front_debuff_idea_6
					hidden_effect = {
						add_ideas = TPMP_USA_alaska_idea
						add_ideas = TPMP_USA_hawaii_idea

				}
			}
		}
	}

					limit = {
						OR = {
							tag = GER
							is_in_faction_with = GER
							tag = JAP
							is_in_faction_with = JAP
						}
						FROM = {
							OR = {
								tag = USA
								is_in_faction_with = USA
								tag = ENG
								is_in_faction_with = ENG
								tag = SOV
								is_in_faction_with = SOV
							}
						}
					}
					JAP = {
						decrease_pacific_front_debuff_JAP = yes
					}
					USA = {
						increase_pacific_front_debuff_USA = yes
					}
				}
				#USA Takes Japan Held Island
				else_if = {
					limit = {
						OR = {
							tag = USA
							is_in_faction_with = USA
							tag = ENG
							is_in_faction_with = ENG
							tag = SOV
							is_in_faction_with = SOV
						}
						FROM = {
							OR = {
								tag = GER
								is_in_faction_with = GER
								tag = JAP
								is_in_faction_with = JAP
							}
						}
					}
					JAP = {
						increase_pacific_front_debuff_JAP = yes
					}
					USA = {
						decrease_pacific_front_debuff_USA = yes
					}
				}
			}
		}
	}
