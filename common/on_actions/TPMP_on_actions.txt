on_actions = {
	on_startup = {
		effect = {
			set_variable = { TPMP_martial_society_mastery_var = 3.5 }
			set_variable = { TPMP_professional_army_mastery_var = 5.0 }
			set_variable = { TPMP_semi_professional_army_mastery_var = 4.0 }
			set_variable = { TPMP_reservist_system_mastery_var = 2.5 }
			set_variable = { TPMP_militia_system_mastery_var = 1.5 }
			every_country = {
				set_rule = { can_send_volunteers = yes }
			}
			every_country = {
				add_timed_idea = {
					idea = TPMP_starting_efficiency
					days = 1
				}
			}
			GER = {
				set_variable = { TPMP_spanish_civil_war_consumer_var = 0.1 }
				set_variable = { TPMP_spanish_civil_war_volunteer_var = 5 }
			}
		}
	}
	on_monthly = {
		effect = {
			IF = {
				limit = {
					has_idea = martial_society
				}
				add_mastery = {
					amount = 3.5
					folder = land
				}
			}
			ELSE_IF = {
				limit = {
					has_idea = professional_army
				}
				add_mastery = {
					amount = 5.0
					folder = land
				}
			}
			ELSE_IF = {
				limit = {
					has_idea = semi_professional_army
				}
				add_mastery = {
					amount = 4.0
					folder = land
				}
			}
			ELSE_IF = {
				limit = {
					has_idea = reservist_system
				}
				add_mastery = {
					amount = 2.5
					folder = land
				}
			}
			ELSE_IF = {
				limit = {
					has_idea = militia_system
				}
				add_mastery = {
					amount = 1.5
					folder = land
				}
			}
		}
	}
	on_declare_war = {
		effect = {
			if = {
				limit = {
					tag = SPR
					FROM = { tag = SPD }
				}
				#Axis Volunteers (14 Total)
				GER = { add_ideas = TPMP_SPR_volunteers_axis_2 }
				ITA = { add_ideas = TPMP_SPR_volunteers_axis_2 }
				HUN = { add_ideas = TPMP_SPR_volunteers_axis_0 }
				ROM = { add_ideas = TPMP_SPR_volunteers_axis_0 }
				BUL = { add_ideas = TPMP_SPR_volunteers_axis_0 }
				FIN = { add_ideas = TPMP_SPR_volunteers_axis_0 }
				#Allied Volunteers (14 Total)
				SOV = { add_ideas = TPMP_SPR_volunteers_allies_2 }
				UKR = { add_ideas = TPMP_SPR_volunteers_allies_0 }
				ENG = { add_ideas = TPMP_SPR_volunteers_allies_1 }
				FRA = { add_ideas = TPMP_SPR_volunteers_allies_1 }
				CAN = { add_ideas = TPMP_SPR_volunteers_allies_0 }
				RAJ = { add_ideas = TPMP_SPR_volunteers_allies_0 }
				AST = { add_ideas = TPMP_SPR_volunteers_allies_0 }
				SAF = { add_ideas = TPMP_SPR_volunteers_allies_0 }
			}
		}
	}
	on_capitulation = {
		effect = {
			if = {
				limit = {
					OR = {
						tag = SPD
						tag = SPR
					}
					OR = {
						FROM = { tag = SPD }
						FROM = { tag = SPR }
					}
				}
				#Axis Volunteers (14 Total)
				if = {
					limit = {
						GER = { has_idea = TPMP_SPR_volunteers_axis_2 }
					}
					GER = { remove_ideas = TPMP_SPR_volunteers_axis_2 }
				}
				else_if = {
					limit = {
						GER = {
							has_dynamic_modifier = {
								modifier = TPMP_SPR_volunteers_axis_3
							}
						}
					}
					GER = {
						remove_dynamic_modifier = {
							modifier = TPMP_SPR_volunteers_axis_3
						}
					}
				}
				ITA = { remove_ideas = TPMP_SPR_volunteers_axis_2 }
				HUN = { remove_ideas = TPMP_SPR_volunteers_axis_0 }
				ROM = { remove_ideas = TPMP_SPR_volunteers_axis_0 }
				BUL = { remove_ideas = TPMP_SPR_volunteers_axis_0 }
				FIN = { remove_ideas = TPMP_SPR_volunteers_axis_0 }
				#Allied Volunteers (14 Total)
				SOV = { remove_ideas = TPMP_SPR_volunteers_allies_2 }
				UKR = { remove_ideas = TPMP_SPR_volunteers_allies_0 }
				ENG = { remove_ideas = TPMP_SPR_volunteers_allies_1 }
				FRA = { remove_ideas = TPMP_SPR_volunteers_allies_1 }
				CAN = { remove_ideas = TPMP_SPR_volunteers_allies_0 }
				RAJ = { remove_ideas = TPMP_SPR_volunteers_allies_0 }
				AST = { remove_ideas = TPMP_SPR_volunteers_allies_0 }
				SAF = { remove_ideas = TPMP_SPR_volunteers_allies_0 }
			}
		}
	}
	on_capitulation = {
		effect = {
			if = {
				limit = {
					tag = HOL
				}
				AST = {
					country_event = TPMP_AST.1
				}
			}
		}
	}
	on_capitulation_immediate = { #scripted Iran
		effect = {
			if = {
				limit = {
					tag = PER
				}
				ENG = {
					annex_country = {
						target = PER 
						transfer_troops = no
					}
				}
				SOV = {
					country_event = TPMP_AST.2
				}
			}
		}
	}
	on_capitulation_immediate = { #scripted China
		effect = {
			if = {
				limit = {
					tag = CHI
				}
				JAP = {
					annex_country = {
						target = CHI 
						transfer_troops = yes #Whack but if i dont do this china equipment is all gone
					}
				}
				
			}
		}
	}
}
